# Player keyboard/screenshot helper with real scan codes (SendKeys events were dropped by the player).
# Usage: pkey.ps1 -Keys "TAB,TAB,ENTER" [-Shot name] [-Launch build -Width w -Height h] [-Quit]
param([string]$Keys = '', [string]$Shot = '', [string]$Launch = '', [int]$Width = 1920, [int]$Height = 1080, [switch]$Quit, [string]$Tag = 'x', [switch]$FocusOnly)
Add-Type -AssemblyName System.Windows.Forms, System.Drawing
if (-not ('KeyIn2' -as [type])) {
Add-Type @"
using System; using System.Runtime.InteropServices;
public static class KeyIn2 {
  [DllImport("user32.dll")] static extern void keybd_event(byte vk, byte scan, uint flags, UIntPtr extra);
  [DllImport("user32.dll")] static extern uint MapVirtualKey(uint code, uint mapType);
  [DllImport("user32.dll")] public static extern bool SetForegroundWindow(IntPtr h);
  [DllImport("user32.dll")] static extern IntPtr GetForegroundWindow();
  [DllImport("user32.dll")] static extern uint GetWindowThreadProcessId(IntPtr h, out uint pid);
  public static uint ForegroundPid() { uint pid; GetWindowThreadProcessId(GetForegroundWindow(), out pid); return pid; }
  static uint Ext(byte vk) { return (vk >= 0x21 && vk <= 0x2E) ? 1u : 0u; }
  public static void Down(byte vk) { keybd_event(vk, (byte)MapVirtualKey(vk, 0), Ext(vk), UIntPtr.Zero); }
  public static void Up(byte vk) { keybd_event(vk, (byte)MapVirtualKey(vk, 0), Ext(vk) | 2u, UIntPtr.Zero); }
}
"@
}
$vk = @{ TAB = 0x09; ENTER = 0x0D; ESC = 0x1B; LEFT = 0x25; UP = 0x26; RIGHT = 0x27; DOWN = 0x28; SHIFT = 0xA0 }
$perf = 'C:\Users\azymi\Desktop\S0-host-acceptance-20261018\perf'
function Focus { $p = Get-Process SystemXI -ErrorAction Stop | Select-Object -First 1; if ([KeyIn2]::ForegroundPid() -ne $p.Id) { [void][KeyIn2]::SetForegroundWindow($p.MainWindowHandle); Start-Sleep -Milliseconds 250 } }
# Safety: a key is only ever sent while the player owns the foreground window; otherwise abort.
function Guard { $p = Get-Process SystemXI -ErrorAction Stop | Select-Object -First 1; if ([KeyIn2]::ForegroundPid() -ne $p.Id) { Focus; if ([KeyIn2]::ForegroundPid() -ne $p.Id) { throw "ABORT: player is not the foreground window; no key sent" } } }
if ($Launch) {
    Start-Process -FilePath (Join-Path $Launch 'SystemXI.exe') -ArgumentList "-screen-fullscreen 0 -screen-width $Width -screen-height $Height -logFile `"$(Join-Path $Launch 'player.log')`""
    Start-Sleep -Seconds 10
}
if ($FocusOnly) { Focus }
if ($Keys) {
    Focus
    foreach ($k in $Keys.Split(',')) {
        $shift = $k.StartsWith('S-'); $name = $k.TrimStart('S', '-')
        Guard
        if ($shift) { [KeyIn2]::Down($vk.SHIFT); Start-Sleep -Milliseconds 40 }
        [KeyIn2]::Down([byte]$vk[$name]); Start-Sleep -Milliseconds 90; [KeyIn2]::Up([byte]$vk[$name])
        if ($shift) { Start-Sleep -Milliseconds 40; [KeyIn2]::Up($vk.SHIFT) }
        Start-Sleep -Milliseconds 300
    }
}
if ($Shot) {
    Focus
    $b = [System.Windows.Forms.Screen]::PrimaryScreen.Bounds
    $bmp = New-Object System.Drawing.Bitmap $b.Width, $b.Height
    $g = [System.Drawing.Graphics]::FromImage($bmp); $g.CopyFromScreen($b.Location, [System.Drawing.Point]::Empty, $b.Size)
    $path = Join-Path $perf "$Tag-$Shot.png"; $bmp.Save($path); $g.Dispose(); $bmp.Dispose(); "shot $path"
}
if ($Quit) { Get-Process SystemXI -ErrorAction SilentlyContinue | ForEach-Object { $_.CloseMainWindow() | Out-Null }; 'closed' }
