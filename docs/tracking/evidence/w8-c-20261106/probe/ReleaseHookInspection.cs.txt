using System;
using System.IO;
using System.Linq;
using System.Reflection.Metadata;
using System.Reflection.Metadata.Ecma335;
using System.Reflection.PortableExecutable;

foreach (string path in args)
{
    bool debug = path.Contains("/Debug/");
    using var stream = File.OpenRead(path);
    using var pe = new PEReader(stream);
    MetadataReader reader = pe.GetMetadataReader();
    var type = reader.GetTypeDefinition(reader.TypeDefinitions.Single(h =>
        reader.GetString(reader.GetTypeDefinition(h).Name) == "MatchEngine"));
    string[] fields = type.GetFields().Select(h => reader.GetString(reader.GetFieldDefinition(h).Name)).ToArray();
    foreach (string field in new[] { "TestOnly_PostReattachmentBall", "TestOnly_KeeperWhistleFinishCount" })
    {
        bool present = fields.Any(name => name.Contains(field));
        if (present != debug) throw new Exception($"Unexpected field {field}: present={present} in {path}");
    }
    foreach (var pair in new[] {
        (caller: "RunResolvePhase", callee: "ApplyHandControlFixtureState"),
        (caller: "FinishKeeperWhistle", callee: "RecordKeeperWhistleForTests") })
    {
        var caller = type.GetMethods().Single(h => reader.GetString(reader.GetMethodDefinition(h).Name) == pair.caller);
        var callee = type.GetMethods().Single(h => reader.GetString(reader.GetMethodDefinition(h).Name) == pair.callee);
        byte[] il = pe.GetMethodBody(reader.GetMethodDefinition(caller).RelativeVirtualAddress).GetILBytes();
        byte[] token = BitConverter.GetBytes(MetadataTokens.GetToken(callee));
        int calls = 0;
        for (int i = 0; i + 4 < il.Length; i++)
            if (il[i] == 0x28 && il.AsSpan(i + 1, 4).SequenceEqual(token)) calls++;
        if (calls != (debug ? 1 : 0)) throw new Exception($"Unexpected {pair.caller}->{pair.callee} call count: {calls}");
        Console.WriteLine($"{(debug ? "Development" : "Release")}: {pair.caller}->{pair.callee} calls={calls}");
    }
    Console.WriteLine($"{(debug ? "Development" : "Release")}: fixture field presence is correct.");
}
