param([string]$Player, [string]$Product, [string]$Mode, [string]$Tag, [int]$Width = 1366, [int]$Height = 768, [int]$TimeoutSeconds = 1800)
# Launches one development probe player in one mode, waits for it to exit, and snapshots its save directory.
$ErrorActionPreference = 'Stop'
$S = '<scratch>'
$P = Join-Path $S ("run2-" + $Product.Replace(' ', '-'))
$C = Join-Path $env:USERPROFILE ("AppData\LocalLow\DefaultCompany\" + $Product + "\careers")
New-Item -ItemType Directory -Force $P | Out-Null
function Note($s) { $line = (Get-Date).ToUniversalTime().ToString('o') + ' ' + $s; Add-Content -Path (Join-Path $P 'harness.log') -Value $line; $line }

$argv = @('-careerAccept', $Mode, '-probeDir', $P, '-screen-fullscreen', '0', '-screen-width', "$Width", '-screen-height', "$Height", '-logFile', (Join-Path $P "player-$Tag.log"))
$sw = [Diagnostics.Stopwatch]::StartNew()
$proc = Start-Process -FilePath $Player -ArgumentList $argv -PassThru
Note "launched $Mode ($Tag) pid=$($proc.Id) ${Width}x${Height}"
$exited = $proc.WaitForExit($TimeoutSeconds * 1000)
if (-not $exited) { Note "TIMEOUT ($Tag); killing"; $proc.Kill(); $proc.WaitForExit() }
Note ("exited ($Tag) after " + [math]::Round($sw.Elapsed.TotalSeconds, 1) + " s exitCode=" + $proc.ExitCode)
$dst = Join-Path $P ("snap-" + $Tag)
if (Test-Path $dst) { Remove-Item -Recurse -Force $dst }
if (Test-Path $C) {
    Copy-Item -Recurse $C $dst
    Get-ChildItem $dst | ForEach-Object { Note ("  " + $_.Name + " " + $_.Length + " " + (Get-FileHash $_.FullName -Algorithm SHA256).Hash.Substring(0, 16)) }
}
