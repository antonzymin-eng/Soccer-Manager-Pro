        run: |
          set -euo pipefail
          OUT="$RUNNER_TEMP/w2-442-base"
          mkdir -p "$OUT"

          FILTER='Name=TacklesHappenInComposedPlay|Name=AControlledCarrierIsActuallyDispossessed|Name=BothOutcomesOccur_TheBallIsSometimesWonAndSometimesKnockedLoose|Name=ChallengesAreBoundedAbove_NotJustNonZero|Name=TheCooldownArmsOnEveryOutcomeIncludingAMiss|Name=ChallengesOccurForBothDefendingTeams|Name=ATackleFoulIsGivenAsASlideTackleAndNotJudgedTwice|Name=ATackleFoulIsPublishedAsASlideTackle|Name=ReportW2442Corpus'

          patch_reporter() {
            python3 - <<'PY'
          from pathlib import Path

          path = Path("src/match-engine/tests/MatchEngineTackleTests.cs")
          text = path.read_text(encoding="utf-8")
          needle = "        [Test]\n        public void SaveAndRestoreCarryTheTackleLatches"
          reporter = """        [Test]
                  public void ReportW2442Corpus()
                  {
                      int connected = s_pooled.Won + s_pooled.Loose + s_pooled.Foul;
                      TestContext.WriteLine(
                          "W2_442_COUNTS won=" + s_pooled.Won
                          + " loose=" + s_pooled.Loose
                          + " foul=" + s_pooled.Foul
                          + " missed=" + s_pooled.Missed
                          + " dispossessions=" + s_dispossessions);
                      TestContext.WriteLine(
                          "W2_442_GUARDS connected=" + connected
                          + " foul=" + s_pooled.Foul
                          + " team0Resolved=" + s_perTeamResolved[0]
                          + " team1Resolved=" + s_perTeamResolved[1]
                          + " slideTackleFouls=" + s_slideTackleFouls
                          + " cooldownOnMissOnly=" + s_sawCooldownWhileOnlyMisses);
                  }

          """
          if text.count(needle) != 1:
              raise SystemExit("reporter insertion point not unique")
          path.write_text(text.replace(needle, reporter + needle, 1), encoding="utf-8", newline="\n")
          PY
          }

          run_arm() {
            label="$1"
            ref="$2"
            expected_blob="$3"

            git reset --hard
            git clean -fdx
            git checkout --detach "$ref"
            test "$(git rev-parse HEAD)" = "$ref"
            test "$(git hash-object src/match-engine/tests/MatchEngineTackleTests.cs)" = "$expected_blob"

            mkdir -p "$OUT/$label"
            {
              echo "label=$label"
              echo "ref=$ref"
              echo "pre_report_test_blob=$expected_blob"
              echo "runner_os=$RUNNER_OS"
              echo "runner_arch=$RUNNER_ARCH"
            } > "$OUT/$label/provenance.txt"

            patch_reporter
            git diff --check -- src/match-engine/tests/MatchEngineTackleTests.cs
            git diff -- src/match-engine/tests/MatchEngineTackleTests.cs > "$OUT/$label/reporter.patch"

            python3 tools/dotnet-ci/generate_projects.py
            dotnet restore tools/dotnet-ci/TacticalDirector.gen.sln
            dotnet build src/match-engine/tests/match-engine-tests.gen.csproj --no-restore

            set +e
            /usr/bin/time -f 'elapsed_seconds=%e' -o "$OUT/$label/runtime.txt" \
              bash -o pipefail -c "dotnet test src/match-engine/tests/match-engine-tests.gen.csproj --no-build --no-restore --filter \"$FILTER\" --logger \"console;verbosity=detailed\" --logger \"trx;LogFileName=$label.trx\" --results-directory \"$OUT/$label\" 2>&1 | tee \"$OUT/$label/console.log\""
            rc=$?
            set -e
            echo "test_exit_code=$rc" >> "$OUT/$label/provenance.txt"
            grep -E 'W2_442_(COUNTS|GUARDS)' "$OUT/$label/console.log" || true
          }

          run_arm base-old "$BASE_OLD_SHA" "$BASE_OLD_TEST_BLOB"
          run_arm base-contract "$BASE_CONTRACT_SHA" "$CONTRACT_TEST_BLOB"

          echo "=== runtime comparison ==="
          cat "$OUT/base-old/runtime.txt"
          cat "$OUT/base-contract/runtime.txt"

      - name: Upload base evidence
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: w2-442-base-${{ github.run_id }}
          path: ${{ runner.temp }}/w2-442-base/
          if-no-files-found: error
