// Scratch host-validation build step. Never committed.
// -executeMethod BuildProbePlayer.Build -probeOut <exe path>: Win64 development player of the build-settings scenes.
using System;
using UnityEditor;
using UnityEditor.Build.Reporting;

public static class BuildProbePlayer
{
    public static void Build()
    {
        string output = null;
        string[] args = Environment.GetCommandLineArgs();
        for (int i = 0; i < args.Length - 1; i++)
            if (args[i] == "-probeOut") output = args[i + 1];
        var scenes = new System.Collections.Generic.List<string>();
        foreach (EditorBuildSettingsScene scene in EditorBuildSettings.scenes)
            if (scene.enabled) scenes.Add(scene.path);
        BuildReport report = BuildPipeline.BuildPlayer(new BuildPlayerOptions
        {
            scenes = scenes.ToArray(),
            locationPathName = output,
            target = BuildTarget.StandaloneWindows64,
            options = BuildOptions.Development,
        });
        UnityEngine.Debug.Log("[BuildProbePlayer] result=" + report.summary.result + " scenes=" + string.Join(",", scenes)
            + " errors=" + report.summary.totalErrors + " size=" + report.summary.totalSize);
        EditorApplication.Exit(report.summary.result == BuildResult.Succeeded ? 0 : 1);
    }
}
