import sys,struct
def f(h): return struct.unpack('>f',bytes.fromhex(h))[0]
for p in sys.argv[1:]:
    rows=[]
    for l in open(p):
        if l.startswith('tick'): continue
        t,b,_=l.split('\t'); t=int(t)
        if t<3600 or t%600: continue
        rows.append((t,f(b[0:8]),f(b[8:16]),f(b[24:32]),f(b[32:40]),b[48:]))
    ft=sum(1 for r in rows if r[1]<35 or r[1]>70)
    # longest run of consecutive checkpoints with ball within 1 m of previous
    best=(0,0);cur=0;start=0
    for i in range(1,len(rows)):
        if abs(rows[i][1]-rows[i-1][1])<1 and abs(rows[i][2]-rows[i-1][2])<1:
            if cur==0: start=rows[i-1][0]
            cur+=1
            if cur>best[0]: best=(cur,start)
        else: cur=0
    r0=[r for r in rows if r[0]==best[1]]
    print(p.split('trace-')[1][:18], 'checkpoints',len(rows),'final-third',ft, 'longest static run',best[0]*10,'s from tick',best[1], 'ball',r0[0][1:3] if r0 else None,'state',r0[0][5] if r0 else None)
